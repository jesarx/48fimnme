% el texto se achica (de 2 en 2 %) hasta #1 mientras no quepa con la franja
\newcommand{\achicarhasta}[1]{%
  \loop
    \achicarfalse
    \ifdim\dimexpr\altocontenido+\separacion+\franjamin\relax>\zona
      \ifdim\escala pt>\fpeval{#1+0.01}pt \achicartrue\fi\fi
  \ifachicar
    \edef\escala{\fpeval{\escala-0.02}}\armar
  \repeat}
% la franja baja hasta #1 (si no es ya más baja) mientras el texto no quepa
\newcommand{\bajarfranja}[1]{%
  \ifx\fotoslua\empty\else
    \ifdim\dimexpr\altocontenido+\separacion+\franjamin\relax>\zona
      \setlength{\franjamin}{\dimexpr\zona-\separacion-\altocontenido\relax}
      \ifdim\franjamin<#1\relax \setlength{\franjamin}{#1}\fi
    \fi
  \fi}
